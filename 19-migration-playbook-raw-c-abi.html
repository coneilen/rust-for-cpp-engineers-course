// legacy-cpp/alert_scoring.cpp
#include "alert_scoring.h"

#include <algorithm>
#include <cmath>
#include <deque>
#include <utility>

class AlertScorer::Impl {
public:
    Impl(double baselineWindowSeconds, double sensitivity)
        : baseline_window_ms_(static_cast<std::int64_t>(
              baselineWindowSeconds * 1000.0)),
          sensitivity_(sensitivity) {}

    double Score(const MetricSample& sample) {
        if (!std::isfinite(sample.value)) {
            throw ScoringError("sample value must be finite");
        }

        while (!window_.empty() &&
               sample.timestamp_unix_ms -
                       window_.front().timestamp_unix_ms >
                   baseline_window_ms_) {
            window_.pop_front();
        }

        if (window_.empty()) {
            window_.push_back(sample);
            throw ScoringError("baseline window is empty");
        }

        double sum = 0.0;
        for (const MetricSample& item : window_) {
            sum += item.value;
        }
        const double mean = sum / static_cast<double>(window_.size());

        double squared_error = 0.0;
        for (const MetricSample& item : window_) {
            const double delta = item.value - mean;
            squared_error += delta * delta;
        }
        const double variance =
            squared_error / static_cast<double>(window_.size());
        const double standard_deviation =
            std::max(std::sqrt(variance), 1e-6);
        const double z_score =
            std::abs((sample.value - mean) / standard_deviation);
        const double score =
            std::clamp(z_score * sensitivity_, 0.0, 1.0);

        window_.push_back(sample);
        return score;
    }

    void Reset() { window_.clear(); }

private:
    std::deque<MetricSample> window_;
    std::int64_t baseline_window_ms_;
    double sensitivity_;
};

AlertScorer::AlertScorer(
    double baselineWindowSeconds,
    double sensitivity) {
    if (!std::isfinite(baselineWindowSeconds) ||
        baselineWindowSeconds <= 0.0) {
        throw std::invalid_argument(
            "baseline window must be finite and positive");
    }
    if (!std::isfinite(sensitivity) ||
        sensitivity <= 0.0 || sensitivity > 1.0) {
        throw std::invalid_argument(
            "sensitivity must be in (0.0, 1.0]");
    }
    impl_ = std::make_unique<Impl>(
        baselineWindowSeconds, sensitivity);
}

AlertScorer::~AlertScorer() = default;
AlertScorer::AlertScorer(AlertScorer&&) noexcept = default;
AlertScorer& AlertScorer::operator=(
    AlertScorer&&) noexcept = default;

double AlertScorer::Score(const MetricSample& sample) {
    return impl_->Score(sample);
}

void AlertScorer::Reset() {
    impl_->Reset();
}

A small, stable, well-called seam is the best first slice

AlertScorer is a good first migration target precisely because it has one caller family, a narrow public surface, and no shared mutable state to untangle. Save the tangled, widely-called modules for later, once the playbook has proven itself.

#include <cstdint>
#include <cstdio>
#include <fstream>
#include <sstream>
#include <stdexcept>
#include <string>
#include <vector>

namespace {
struct FixtureRow {
    double value;
    std::int64_t timestamp_unix_ms;
    double expected_score;
};

std::vector<FixtureRow> LoadFixtureRows(const std::string& path) {
    std::ifstream in(path);
    if (!in) {
        throw std::runtime_error("failed to open fixture file: " + path);
    }
    std::vector<FixtureRow> rows;
    std::string line;
    while (std::getline(in, line)) {
        if (line.empty() || line[0] == '#') {
            continue;
        }
        std::stringstream ss(line);
        std::string value_str, ts_str, score_str;
        std::getline(ss, value_str, ',');
        std::getline(ss, ts_str, ',');
        std::getline(ss, score_str, ',');
        rows.push_back(FixtureRow{std::stod(value_str), std::stoll(ts_str), std::stod(score_str)});
    }
    return rows;
}
}  // namespace

int main() {
    AlertScorer scorer(/*baselineWindowSeconds=*/300.0, /*sensitivity=*/0.62);

    // One untracked warm-up sample seeds the baseline window: a fresh
    // scorer's very first call always throws ScoringError (cold-start
    // EmptyWindow), matching Part 3's Rust translation, so the fixture
    // below only records the seven rows scored *after* this warm-up.
    try {
        scorer.Score(MetricSample{100.0, 1758067199000});
        std::fprintf(stderr, "expected the warm-up sample to throw, but it did not\n");
        return 1;
    } catch (const std::exception&) {
        // Expected exactly once, for this warm-up sample only.
    }

    auto rows = LoadFixtureRows("legacy-cpp/testdata/course_fixture.csv");
    for (std::size_t i = 0; i < rows.size(); ++i) {
        double score = scorer.Score(MetricSample{rows[i].value, rows[i].timestamp_unix_ms});
        double diff = std::fabs(score - rows[i].expected_score);
        if (diff >= 1e-6) {
            std::fprintf(stderr, "row %zu: got %.9f, expected %.9f\n", i, score, rows[i].expected_score);
            return 1;
        }
    }
    std::printf("characterization parity ok: %zu/%zu rows matched\n", rows.size(), rows.size());
    return 0;
}

Part 3: choose the seam, translate the invariants into Rust

The seam is the public surface identified in Part 1. Each C++ invariant needs an explicit, idiomatic Rust equivalent — not a literal transliteration:

C++ ruleRust translation
Throws on invalid construction inputResult<Self, ScoringError> instead of a thrown exception
Throws on malformed/non-finite sampleExplicit value.is_finite() guard, returns Err(ScoringError::NonFiniteValue) instead of throwing
Throws ScoringError if the baseline window is emptyNew sample is scored against the existing baseline before joining it; an empty baseline returns Err(ScoringError::EmptyWindow) instead of throwing
Single owner via unique_ptr<Impl>An ordinary owned struct; Rust's default ownership already gives single ownership
Not copyableSimply don't derive Clone; Rust types are non-copyable by default
Not thread-safe, caller already serializesMethods take &mut self, which the compiler enforces as exclusive access — no implicit sharing is even possible

// crates/pulsewatch-scoring/src/lib.rs (replace, complete)
use std::collections::VecDeque;
use std::time::Duration;

pub struct Sample {
    pub value: f64,
    pub timestamp_unix_ms: i64,
}

#[derive(Debug, PartialEq, Eq)]
pub enum ScoringError {
    SensitivityOutOfRange,
    InvalidBaselineWindow,
    EmptyWindow,
    NonFiniteValue,
}

pub struct AlertScorer {
    window: VecDeque<Sample>,
    baseline_window: Duration,
    sensitivity: f32,
}

impl AlertScorer {
    pub fn new(baseline_window: Duration, sensitivity: f32) -> Result<Self, ScoringError> {
        if baseline_window.is_zero() {
            return Err(ScoringError::InvalidBaselineWindow);
        }
        if !(0.0..=1.0).contains(&sensitivity) || sensitivity == 0.0 {
            return Err(ScoringError::SensitivityOutOfRange);
        }
        Ok(Self { window: VecDeque::new(), baseline_window, sensitivity })
    }

    pub fn score(&mut self, sample: Sample) -> Result<f32, ScoringError> {
        if !sample.value.is_finite() {
            return Err(ScoringError::NonFiniteValue);
        }
        self.evict_outside_baseline_window(sample.timestamp_unix_ms);
        if self.window.is_empty() {
            // Nothing to compare against yet: seed the baseline with this
            // sample and report EmptyWindow, matching the C++ contract
            // ("Throws ScoringError if the window is empty") instead of
            // scoring a sample against itself.
            self.window.push_back(sample);
            return Err(ScoringError::EmptyWindow);
        }
        let score = self.compute_deviation_score(&sample);
        self.window.push_back(sample);
        Ok(score)
    }

    pub fn reset(&mut self) {
        self.window.clear();
    }

    // Drops baseline entries older than the incoming sample's timestamp by
    // more than baseline_window, matching the legacy C++ deque-trim
    // behavior exactly. Runs before the incoming sample joins the baseline.
    fn evict_outside_baseline_window(&mut self, newest_ms: i64) {
        let window_ms = self.baseline_window.as_millis() as i64;
        while let Some(oldest) = self.window.front() {
            if newest_ms - oldest.timestamp_unix_ms > window_ms {
                self.window.pop_front();
            } else {
                break;
            }
        }
    }

    // Z-score of the incoming sample against the *existing* baseline
    // window's mean and standard deviation (the sample itself is not yet
    // part of the window), scaled by sensitivity and clamped to [0.0, 1.0]
    // to match the legacy engine's documented output range.
    fn compute_deviation_score(&self, sample: &Sample) -> f32 {
        let n = self.window.len() as f64;
        let mean: f64 = self.window.iter().map(|s| s.value).sum::<f64>() / n;
        let variance: f64 = self
            .window
            .iter()
            .map(|s| (s.value - mean).powi(2))
            .sum::<f64>()
            / n;
        let stddev = variance.sqrt().max(1e-6); // avoid divide-by-zero on a flat window
        let z = ((sample.value - mean) / stddev).abs();
        (z * self.sensitivity as f64).clamp(0.0, 1.0) as f32
    }
}

Part 4: a raw C ABI so the C++ caller can reach the Rust core

The C++ ingestion service still owns the call site during the migration, so Pulsewatch needs a boundary the existing code can call across today. The simplest, most defensible boundary is a hand-written raw C ABI: #[repr(C)] status codes, unsafe extern "C" functions, and an opaque handle that never exposes the Rust struct's field layout to C++. Every one of these functions dereferences a raw pointer whose validity only the caller can guarantee, so each is declared pub unsafe extern "C" fn — the internal null checks make misuse easier to diagnose, but they do not turn raw-pointer FFI into a safe Rust API. File: append this block to the end of the same path/to/pulsewatch/crates/pulsewatch-scoring/src/lib.rs from Part 3 (it extends, not replaces, what you already wrote):

// crates/pulsewatch-scoring/src/lib.rs (append — continues the file from Part 3)
#[repr(C)]
pub enum PwStatus {
    Ok = 0,
    InvalidSensitivity = 1,
    NullHandle = 2,
    Panicked = 3,
    EmptyWindow = 4,
    NonFiniteValue = 5,
    InvalidBaselineWindow = 6,
}

/// Maps every `ScoringError` variant to its wire status code explicitly, so
/// adding a new error variant is a compile error here until it is handled.
fn map_scoring_error(err: ScoringError) -> PwStatus {
    match err {
        ScoringError::SensitivityOutOfRange => PwStatus::InvalidSensitivity,
        ScoringError::InvalidBaselineWindow => PwStatus::InvalidBaselineWindow,
        ScoringError::EmptyWindow => PwStatus::EmptyWindow,
        ScoringError::NonFiniteValue => PwStatus::NonFiniteValue,
    }
}

/// Opaque to C++: only ever seen and passed around as a pointer.
pub struct PwScorer(AlertScorer);

/// # Safety
/// Callers must satisfy every invariant below; the compiler cannot check
/// any of them for an exported symbol like this one:
/// - `out_handle` must be non-null and point to writable memory for one
///   `*mut PwScorer`.
/// - On `PwStatus::Ok` it is set to a freshly allocated handle that must
///   later be passed to exactly one `pw_scorer_free` call.
#[unsafe(no_mangle)]
pub unsafe extern "C" fn pw_scorer_new(
    baseline_window_secs: f64,
    sensitivity: f32,
    out_handle: *mut *mut PwScorer,
) -> PwStatus {
    if out_handle.is_null() {
        return PwStatus::NullHandle;
    }
    // Fallible conversion rejects negative, non-finite, and overflowing
    // values. Reject zero after rounding as well.
    let baseline_window = match Duration::try_from_secs_f64(baseline_window_secs) {
        Ok(duration) if !duration.is_zero() => duration,
        _ => return map_scoring_error(ScoringError::InvalidBaselineWindow),
    };
    let built =
        std::panic::catch_unwind(|| AlertScorer::new(baseline_window, sensitivity));
    match built {
        Ok(Ok(scorer)) => {
            unsafe { *out_handle = Box::into_raw(Box::new(PwScorer(scorer))); }
            PwStatus::Ok
        }
        Ok(Err(e)) => map_scoring_error(e),
        Err(_) => PwStatus::Panicked,
    }
}

/// # Safety
/// Callers must satisfy every invariant below; the compiler cannot check
/// any of them for an exported symbol like this one:
/// - `handle` must be a live pointer previously returned by `pw_scorer_new`
///   that has not yet been passed to `pw_scorer_free`.
/// - `out_score` must be non-null and point to writable memory for one `f32`.
/// - Callers must serialize calls per handle: `AlertScorer::score` takes
///   `&mut self`, so invoking this concurrently on the same handle from two
///   threads is undefined behavior — matching the "not thread-safe"
///   invariant from Part 1, now enforced by the borrow rather than by
///   caller convention alone.
#[unsafe(no_mangle)]
pub unsafe extern "C" fn pw_scorer_score(
    handle: *mut PwScorer,
    value: f64,
    timestamp_unix_ms: i64,
    out_score: *mut f32,
) -> PwStatus {
    if handle.is_null() || out_score.is_null() {
        return PwStatus::NullHandle;
    }
    let scorer = unsafe { &mut *handle };
    let scored = std::panic::catch_unwind(std::panic::AssertUnwindSafe(|| {
        scorer.0.score(Sample { value, timestamp_unix_ms })
    }));
    match scored {
        Ok(Ok(score)) => { unsafe { *out_score = score; } PwStatus::Ok }
        Ok(Err(e)) => map_scoring_error(e),
        Err(_) => PwStatus::Panicked,
    }
}

/// # Safety
/// `handle` must be null, or a pointer previously returned by
/// `pw_scorer_new` that has not already been freed. Freeing the same
/// handle twice, or a pointer this crate did not allocate, is undefined
/// behavior the compiler cannot check for an exported symbol.
#[unsafe(no_mangle)]
pub unsafe extern "C" fn pw_scorer_free(handle: *mut PwScorer) {
    if handle.is_null() {
        return;
    }
    unsafe { drop(Box::from_raw(handle)); }
}
New syntaxWhy it exists
#[repr(C)] on PwStatusGuarantees the C-compatible layout only for this explicitly marked enum, so C++ can read it as a plain int-backed status code.
#[unsafe(no_mangle)] + extern "C"Keeps the symbol name stable and gives the function the platform C calling convention. Edition 2024 requires the unsafe attribute form because exporting a globally named symbol creates obligations the compiler cannot verify, including avoiding symbol-name collisions.
pub unsafe extern "C" fnEvery one of these functions dereferences or writes through a raw pointer whose validity only the caller can guarantee — that makes them unsafe fn by Rust's own rule (a function is only safe to mark non-unsafe if it is safe to call for any argument values the type system allows). The internal null checks reduce, but do not eliminate, what a misbehaving caller can still get wrong (dangling pointers, wrong handle type, use-after-free), so these stay unsafe entry points, not safe Rust APIs with a safe facade.
Null checks on every raw pointer (out_handle, handle, out_score)Each pointer is validated before it is ever dereferenced. A null value returns PwStatus::NullHandle instead of the undefined behavior a stray dereference would cause — this narrows, but does not remove, the unsafe fn contract above.
/// # Safety doc commentsDocuments the caller obligations the compiler cannot check for an exported unsafe fn — pointer provenance, freeing exactly once, and not calling concurrently on the same handle — matching the convention rustc's own lints expect on public unsafe fns.
PwScorer(AlertScorer) as an opaque handleC++ only ever holds a pointer; it never sees or relies on AlertScorer's internal fields or layout.
Box::into_raw / Box::from_rawHands ownership to C++ as a raw pointer, then reclaims and drops it exactly once in pw_scorer_free — never call C's free() on this pointer.
catch_unwind / AssertUnwindSafeCatches ordinary unwinding Rust panics before they reach the non-unwinding extern "C" boundary, where Rust would otherwise abort the process, and converts them into PwStatus::Panicked. This only works if the crate keeps the default panic = "unwind" strategy; a crate built with panic = "abort" aborts immediately and catch_unwind never runs.
⚠️
Rust's ABI and struct layout are not a stable, foreign-facing contract

Default (repr(Rust)) layout is unspecified and can change between compiler versions or even between builds. Only types explicitly marked #[repr(C)] get a defined, C-compatible layout, and only for the fields you mark. The opaque-handle pattern above sidesteps needing a stable Rust ABI at all: C++ never sees AlertScorer's fields, only an opaque pointer, so both sides must simply be rebuilt together, not independently versioned like a real stable ABI.

A hand-written header must match these declarations exactly. For anything beyond a handful of functions, generate it instead of transcribing it by hand — cbindgen reads the Rust source and emits a matching C or C++ header automatically, so the header can never silently drift from the real signatures.

struct FixtureRow {

// Minimal fixed-column reader for this course's own three-field fixture
// format (value,timestamp_unix_ms,expected_score) -- not a general CSV
// parser, deliberately, so this test pulls in no new dependency.
fn load_fixture_rows(path: &std::path::Path) -> Vec<FixtureRow> {
    let text = std::fs::read_to_string(path)
        .unwrap_or_else(|e| panic!("failed to read {}: {e}", path.display()));
    text.lines()
        .filter(|line| !line.trim().is_empty() && !line.starts_with('#'))
        .map(|line| {

            let expected_score: f32 = fields.next().unwrap().trim().parse().unwrap();
            FixtureRow { value, timestamp_unix_ms, expected_score }
        })
        .collect()
}

// legacy-cpp/rust_backed_scorer.cc (new file, complete)
#include "rust_backed_scorer.h"

#include <stdexcept>

namespace {
[[noreturn]] void ThrowForStatus(PwStatus status) {
    switch (status) {
        case PwStatus::EmptyWindow:
            throw std::runtime_error("RustBackedScorer: EmptyWindow");
        case PwStatus::NonFiniteValue:
            throw std::runtime_error("RustBackedScorer: NonFiniteValue");
        case PwStatus::InvalidSensitivity:
            throw std::runtime_error("RustBackedScorer: InvalidSensitivity");
        case PwStatus::InvalidBaselineWindow:
            throw std::runtime_error("RustBackedScorer: InvalidBaselineWindow");
        case PwStatus::NullHandle:
            throw std::runtime_error("RustBackedScorer: NullHandle");
        case PwStatus::Panicked:
            throw std::runtime_error("RustBackedScorer: the Rust scorer panicked");
        default:
            throw std::runtime_error("RustBackedScorer: unrecognized PwStatus");
    }
}
}  // namespace

RustBackedScorer::RustBackedScorer(double baselineWindowSeconds, double sensitivity) {
    PwStatus status = pw_scorer_new(baselineWindowSeconds, static_cast<float>(sensitivity), &handle_);
    if (status != PwStatus::Ok) {
        ThrowForStatus(status);
    }
}

RustBackedScorer::~RustBackedScorer() {
    pw_scorer_free(handle_);
}

double RustBackedScorer::Score(double value, std::int64_t timestamp_unix_ms) {
    float score = 0.0f;
    PwStatus status = pw_scorer_score(handle_, value, timestamp_unix_ms, &score);
    if (status != PwStatus::Ok) {
        ThrowForStatus(status);
    }
    return score;
}

struct FixtureRow {

std::vector<FixtureRow> LoadFixtureRows(const std::string& path) {
    std::ifstream in(path);
    if (!in) {
        throw std::runtime_error("failed to open fixture file: " + path);
    }
    std::vector<FixtureRow> rows;
    std::string line;
    while (std::getline(in, line)) {
        if (line.empty() || line[0] == '#') {
            continue;
        }
        std::stringstream ss(line);
        std::string value_str, ts_str, score_str;  // score_str is unused: this is a live differential check, not a replay against a recorded value
        std::getline(ss, value_str, ',');
        std::getline(ss, ts_str, ',');
        std::getline(ss, score_str, ',');
        rows.push_back(FixtureRow{std::stod(value_str), std::stoll(ts_str)});
    }
    return rows;
}
}  // namespace

int main() {
    AlertScorer legacy(/*baselineWindowSeconds=*/300.0, /*sensitivity=*/0.62);
    RustBackedScorer rust_backed(/*baselineWindowSeconds=*/300.0, /*sensitivity=*/0.62);

    // baseline windows start in the same state.

  1. Enable use_rust_scorer for a single canary shard while both paths run and their scores are compared live; alert on any divergence beyond the recorded tolerance.

Knowledge check

1. Why write characterization tests before refactoring AlertScorer at all?

Answer: Legacy code's actual, possibly-undocumented behavior is the real specification. Characterization tests pin that behavior over recorded traffic so the Rust migration is checked against reality, not assumptions about what the code was supposed to do.

2. Why does pw_scorer_score return a PwStatus code instead of letting a panic or a Rust Result cross the boundary directly?

Answer: C++ has no ABI representation for Rust's Result, and a Rust panic reaching a non-unwinding extern "C" boundary aborts the process. Catching ordinary unwinding panics before the boundary and returning a plain status gives C++ a defined, checkable outcome. This only works when the crate keeps the default panic = "unwind" strategy — a crate built with panic = "abort" aborts immediately and catch_unwind never runs.

3. Does marking PwStatus as #[repr(C)] make AlertScorer's internal fields part of a stable ABI?

Answer: No. Only PwStatus itself gets a defined C-compatible layout. AlertScorer stays entirely behind the opaque PwScorer pointer, so its internal layout can change freely as long as both sides are rebuilt together.

4. Why is pw_scorer_score declared pub unsafe extern "C" fn instead of plain pub extern "C" fn, given that it already checks its pointers for null?

Answer: A function is only safe to mark non-unsafe if it is sound for every input the type system allows. Null checks rule out one failure mode, but the compiler still cannot verify pointer provenance, that handle came from pw_scorer_new, or that no other thread is calling it concurrently — those remain caller obligations, so the function stays unsafe fn with a documented /// # Safety contract rather than a safe API with a safe-looking facade.

5. What makes the staged rollout reversible without a redeploy?

Answer: Both the legacy C++ path and the new Rust path stay compiled into the running binary during the transition. Rolling back is a runtime configuration flag flip, not a code revert or rebuild.

Recap and decision rules

Legacy behavior undocumented?      Write characterization tests first
Public surface small and stable?   Good seam for the first migration slice
Crossing to a non-Rust caller?     Start with #[repr(C)] + opaque handles
Exported fn takes a raw pointer?   Mark it pub unsafe extern "C" fn, not just extern "C"
Every raw pointer at the boundary? Validate for null before every dereference
Need to survive a caller's panic?  Wrap every extern "C" entry point in catch_unwind, keep panic="unwind"
Comparing implementations?         Run differential tests over recorded production traffic
Not fully confident yet?           Ship behind a runtime flag with an explicit rollback path
